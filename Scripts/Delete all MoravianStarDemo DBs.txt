-- Delete all MoravianStarDemo_ DBs:

DECLARE @databaseName NVARCHAR(100)
DECLARE db_cursor CURSOR FOR
SELECT name
FROM master.dbo.sysdatabases
WHERE name LIKE 'MoravianStarDemo_%'

OPEN db_cursor
FETCH NEXT FROM db_cursor INTO @databaseName

WHILE @@FETCH_STATUS = 0
BEGIN
    DECLARE @sql NVARCHAR(100)
	SET @sql = 'ALTER DATABASE ' + QUOTENAME(@databaseName) + ' SET SINGLE_USER WITH ROLLBACK IMMEDIATE'
	EXEC(@sql)

    SET @sql = 'USE master; DROP DATABASE ' + QUOTENAME(@databaseName)
    EXEC(@sql)

    FETCH NEXT FROM db_cursor INTO @databaseName
END

CLOSE db_cursor
DEALLOCATE db_cursor
